Kirby Game

Python 


from random import randint

WIDTH = 400
HEIGHT = 400

score = 0
game_over = False
time_left = 20

kirby = Actor("kirby")
kirby.pos = 100, 100

coin = Actor("coin")
coin.pos = 200, 200


def draw():
    screen.fill("green")

    if not game_over:
        kirby.draw()
        coin.draw()

        screen.draw.text(
            "Score: " + str(score),
            color="black",
            topleft=(10, 10)
        )

        screen.draw.text(
            "Time: " + str(time_left),
            color="black",
            topright=(390, 10)
        )

    else:
        screen.fill("pink")

        screen.draw.text(
            "Final Score: " + str(score),
            center=(200, 160),
            fontsize=45,
            color="black"
        )

        screen.draw.text(
            "Press R to Restart",
            center=(200, 230),
            fontsize=30,
            color="black"
        )


def place_coin():
    coin.x = randint(20, WIDTH - 20)
    coin.y = randint(20, HEIGHT - 20)


def countdown():
    global time_left
    global game_over

    if not game_over:
        time_left -= 1

        if time_left <= 0:
            time_left = 0
            game_over = True


def restart_game():
    global score
    global game_over
    global time_left

    score = 0
    game_over = False
    time_left = 20

    kirby.pos = 100, 100
    place_coin()


def update():
    global score

    if game_over:
        if keyboard.r:
            restart_game()

        return


    # Kirby gets faster as score increases
    speed = 2 + (score // 5)


    # Separate if statements allow diagonal movement
    if keyboard.left:
        kirby.x -= speed

    if keyboard.right:
        kirby.x += speed

    if keyboard.up:
        kirby.y -= speed

    if keyboard.down:
        kirby.y += speed


    # Keep Kirby inside the screen
    kirby.x = max(20, min(WIDTH - 20, kirby.x))
    kirby.y = max(20, min(HEIGHT - 20, kirby.y))


    # Coin collision
    if kirby.colliderect(coin):
        score += 1

        sounds.coin.play()

        place_coin()


clock.schedule_interval(countdown, 1.0)
)
